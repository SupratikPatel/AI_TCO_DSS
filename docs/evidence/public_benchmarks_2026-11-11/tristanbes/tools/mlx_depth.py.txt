"""MLX long-context run with a quantized KV cache and a measured TTFT (mlx-lm 0.32.x).

mlx_lm.benchmark has no KV-bits option; this mirrors it (random tokens, seed 0, EOS disabled,
one warm-up) but passes kv_bits / kv_group_size / quantized_kv_start to stream_generate and
times the first token on the wall clock. Prints one JSON line (consumed by bench.py).

    uvx --from mlx-lm==0.32.2 python tools/mlx_depth.py --model <dir> -p 32768 -g 128 -n 3 --kv-bits 8
"""
import argparse, json, statistics, time
import mlx.core as mx
from mlx_lm import load, stream_generate

ap = argparse.ArgumentParser()
ap.add_argument("--model", required=True)
ap.add_argument("-p", "--prompt-tokens", type=int, default=4096)
ap.add_argument("-g", "--generation-tokens", type=int, default=128)
ap.add_argument("-n", "--num-trials", type=int, default=3)
ap.add_argument("--kv-bits", type=int, default=0, help="0 = no KV quantization")
ap.add_argument("--kv-group-size", type=int, default=64)
ap.add_argument("--quantized-kv-start", type=int, default=0, help="quantize from the first token")
ap.add_argument("--prefill-step-size", type=int, default=2048)
ap.add_argument("--delay", type=int, default=0)
a = ap.parse_args()

mx.random.seed(0)
model, tokenizer, config = load(a.model, return_config=True, tokenizer_config={"trust_remote_code": True})
tokenizer._eos_token_ids = {}          # same trick as mlx_lm.benchmark: never stop early
vocab = config.get("vocab_size") or config["text_config"]["vocab_size"]
prompt = mx.random.randint(0, vocab, (a.prompt_tokens,)).tolist()
kw = dict(max_tokens=a.generation_tokens, prefill_step_size=a.prefill_step_size)
if a.kv_bits:
    kw.update(kv_bits=a.kv_bits, kv_group_size=a.kv_group_size, quantized_kv_start=a.quantized_kv_start)


def once():
    t0 = time.perf_counter()
    ttft = None
    for r in stream_generate(model, tokenizer, prompt, **kw):
        if ttft is None:
            ttft = time.perf_counter() - t0
    return r, ttft, time.perf_counter() - t0


once()  # warm-up
pp, tg, ttfts, mem = [], [], [], []
for _ in range(a.num_trials):
    if a.delay:
        time.sleep(a.delay)
    mx.reset_peak_memory()
    r, ttft, total = once()
    pp.append(r.prompt_tps); tg.append(r.generation_tps); ttfts.append(ttft); mem.append(r.peak_memory)

# mean of the trials, as mlx_lm.benchmark's Averages line and llama-bench's avg_ts; median kept alongside
mean = lambda xs: round(statistics.mean(xs), 3)
med = lambda xs: round(statistics.median(xs), 3)
sd = lambda xs: round(statistics.stdev(xs), 3) if len(xs) > 1 else None
print(json.dumps({"prompt_tokens": a.prompt_tokens, "depth": a.prompt_tokens,
                  "kv_type": f"mlx q{a.kv_bits} g{a.kv_group_size}" if a.kv_bits else "f16",
                  "prefill_tps": mean(pp), "prefill_tps_median": med(pp), "prefill_tps_std": sd(pp),
                  "decode_tps": mean(tg), "decode_tps_median": med(tg), "decode_tps_std": sd(tg),
                  "ttft_s": mean(ttfts), "ttft_s_median": med(ttfts), "ttft_method": "measured wall clock to first streamed token",
                  "peak_memory_gb": round(max(mem), 2), "samples_prefill": pp, "samples_decode": tg,
                  "samples_ttft": ttfts, "prefill_step_size": a.prefill_step_size}))
